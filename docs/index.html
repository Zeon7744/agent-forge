快速开始

# 安装
pip install agent-forge

# 创建新 Skill
forge create my-skill
cd my-skill

# 开发工具
python -c "from agent_forge import ToolBase; ..."

# 测试运行
forge test

# 发布到注册表
forge publish

# 启动 MCP 服务器
forge serve --port 8000
